-- which cities spend the most?
SELECT c.city,
       COUNT(i.invoice_id) AS invoices,
       SUM(i.total)        AS revenue
FROM customer c
JOIN invoice i
  ON i.customer_id = c.customer_id
GROUP BY c.city
ORDER BY revenue DESC
LIMIT 10;
